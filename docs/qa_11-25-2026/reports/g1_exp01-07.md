    p, d = k/n, 1 + z*z/n
    c = (p + z*z/(2*n))/d
    h = z*np.sqrt(p*(1-p)/n + z*z/(4*n*n))/d
    return c-h, c+h

for f in (exp(5)/"expt_v1/results").glob("per_task_*.csv"):
