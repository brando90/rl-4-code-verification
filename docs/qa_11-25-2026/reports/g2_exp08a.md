from pathlib import Path
from collections import Counter
import csv, json, numpy as np
B = next(Path('experiments').glob('08_*'))
def rows(p):
    with p.open() as f: return list(csv.DictReader(f))
def jsonl(p):
    return [json.loads(s) for s in p.read_text().splitlines()]

# Count the disputed error class, once per scored task/run.
a = jsonl(B/'expt_v1/results/scored_cells.jsonl')
for run in ('s1', 's2', 's3', 's4'):
    q = [r for r in a if r['teacher']=='opus55' and r['r']==run]
    print(run, len(q), sum(r['error_type']=='MaxOutputTokensExceededError'
                          for r in q))  # 539 each; 79, 78, 74, 78

# Validation: preserve eight attempts per task in each bootstrap draw.
ids = (B/'expt_v1/results/split/vbv1_validation_task_ids.txt').read_text().splitlines()
draw = np.random.default_rng(0).integers(0, 135, (10000, 135))
def validation(p):
    a = rows(p); key = 'kernel_binary' if 'kernel_binary' in a[0] else 'binary_success'
    ix = {(r['task_id'], int(r['sample_idx'])): r for r in a}
    assert len(a) == len(ix) == 1080
    x = np.array([[float(ix[t,k][key]) for k in range(8)] for t in ids])
    v = x.mean(1); mixed = int(((v>0)&(v<1)).sum())
    ex = sum(r['extracted']=='True' for r in a)
    cap = sum(r['capped']=='True' for r in a)
    print(p.name, int(x.sum()), mixed, ex, cap, v.mean(),
          np.quantile(v[draw].mean(1), [.025,.975]),
          ex>=972 and cap<=108 and mixed>=7)
    return v
for version in (1, 3):
    for p in sorted((B/f'expt_v{version}/results/ladder').glob('*_validation_cells.csv')):
        validation(p)
control = validation(B/'expt_v6/results/analysis/1p7_control_cells.csv')
mixed = validation(B/'expt_v6/results/analysis/1p7_mixed_cells.csv')
current = validation(B/'expt_v12/results/analysis_v12_current_cells.csv')
for d in (mixed-control, current-control, current-mixed):
    print('paired', d.mean(), np.quantile(d[draw].mean(1), [.025,.975]))

# Check the summaries' exact factor-equality assertion.
for p in sorted((B/'expt_v1/results/ladder').glob('sft_*_validation_cells.csv')):
    for line, r in enumerate(rows(p), 2):
        if any(float(r[k]) != float(r['kernel_binary'])
               for k in ('IC_tests', 'IC_proofs')):
            print('factor mismatch', p.name, line, r['task_id'], r['sample_idx'])

# Distinguish status labels from final source/replacement outcomes.
a = rows(B/'expt_v13/results/final_row_results.csv')
print(len(a), Counter(r['state'] for r in a))
print(Counter(r['replaced'] for r in a))
# 2585; FALLBACK=1481, DONE=877, UNCHANGED=227; False=1767, True=818
